cmake_minimum_required(VERSION 3.16)
project(humbug_ui LANGUAGES C)
if(NOT TARGET lvgl::lvgl AND NOT TARGET lvgl)
    message(FATAL_ERROR "Add the LVGL target before adding the Cardputer Zero view")
endif()
add_library(humbug_ui STATIC src/humbug_ui.c src/humbug_art.c src/humbug_fonts.c)
target_compile_features(humbug_ui PUBLIC c_std_11)
target_include_directories(humbug_ui PUBLIC include)
if(TARGET lvgl::lvgl)
    target_link_libraries(humbug_ui PUBLIC lvgl::lvgl)
else()
    target_link_libraries(humbug_ui PUBLIC lvgl)
endif()
